        function renderTheoryPage(page) {
            let container = document.getElementById('theory-list');
            container.innerHTML = '';
            
            let totalPages = Math.ceil(shuffledTheoryArray.length / theoryItemsPerPage);
            if (totalPages === 0) totalPages = 1;
            
            let startIndex = (page - 1) * theoryItemsPerPage;
            let endIndex = Math.min(startIndex + theoryItemsPerPage, shuffledTheoryArray.length);
            
            let pageItems = shuffledTheoryArray.slice(startIndex, endIndex);
            
            pageItems.forEach((item, idx) => {
                let globalIdx = startIndex + idx; 
                let cleanQ = item.q.replace(/^\d+\.\s*/, '');
                let div = document.createElement('div');
                div.className = "bg-white border border-slate-200 rounded-lg overflow-hidden";
                div.innerHTML = `
                    <button onclick="document.getElementById('th-rand-${globalIdx}').classList.toggle('hidden')" class="w-full text-left p-4 font-semibold text-slate-800 hover:bg-slate-50 flex justify-between items-center gap-3">
                        <span>${cleanQ}</span><i class="fa-solid fa-chevron-down text-slate-400 flex-none"></i>
                    </button>
                    <div id="th-rand-${globalIdx}" class="hidden p-4 pt-0 text-sm text-slate-700 leading-relaxed border-t">${item.a}</div>
                `;
                container.appendChild(div);
            });

            document.getElementById('theory-page-info').innerText = `Seite ${page} von ${totalPages}`;
            document.getElementById('theory-prev-btn').disabled = (page === 1);
            document.getElementById('theory-next-btn').disabled = (page === totalPages);
        }

        function initTrainer() {
            let specialMapped = zeynepSpecialQuestions.map((sq, i) => ({
                index: i,
                isCustom: true,
                title: sq.title,
                correct: sq.correct,
                options: sq.options,
                image: sq.image,
                style: sq.style,
                features: sq.features
            }));

            let buildingMapped = buildings.map((b, i) => ({
                index: i + 3,
                isCustom: false,
                data: b
            }));

            let allTrainerItems = [...specialMapped, ...buildingMapped];

            allTrainerItems.forEach((item) => {
                if (!trainerData[item.index]) {
                    trainerData[item.index] = { box: 1 };
                }
            });
            saveTrainerData();
            updateTrainerStats();
            loadTrainerQueue();
        }

            document.getElementById('stat-box3').innerText = b3;
        }

            if(boxName === 'box1') {
                badge.className = "px-3 py-1 rounded-full font-semibold uppercase tracking-wider bg-red-100 text-red-800";
                badge.innerText = "In Wiederholung";
            } else if(boxName === 'box2') {
                badge.className = "px-3 py-1 rounded-full font-semibold uppercase tracking-wider bg-amber-100 text-amber-800";
                badge.innerText = "Verstanden";
            } else {
                badge.className = "px-3 py-1 rounded-full font-semibold uppercase tracking-wider bg-emerald-100 text-emerald-800";
                badge.innerText = "Beherrscht";
            }
            loadTrainerQueue();
        }

        function loadTrainerQueue() {
            activeQueue = [];
            let targetBox = currentFilter === 'box1' ? 1 : currentFilter === 'box2' ? 2 : 3;

            let specialMapped = zeynepSpecialQuestions.map((sq, i) => ({
                index: i,
                isCustom: true,
                title: sq.title,
                correct: sq.correct,
                options: sq.options,
                image: sq.image,
                style: sq.style,
                features: sq.features
            }));

            let buildingMapped = buildings.map((b, i) => ({
                index: i + 3,
                isCustom: false,
                data: b
            }));

            let allItems = [...specialMapped, ...buildingMapped];

            if (targetBox === 1) {
                let others = allItems.filter(x => !x.isCustom && trainerData[x.index] && trainerData[x.index].box === 1).sort(() => Math.random() - 0.5);
                let customs = specialMapped.filter(x => trainerData[x.index] && trainerData[x.index].box === 1);

                activeQueue = [...others];

                let targetPositions = [5, 14, 29];
                customs.forEach((c, idx) => {
                    let pos = targetPositions[idx];
                    if (pos !== undefined) {
                        if (activeQueue.length >= pos) {
                            activeQueue.splice(pos, 0, c);
                        } else {
                            activeQueue.push(c);
                        }
                    }
                });
            } else {
                allItems.forEach(item => {
                    let k = item.index;
                    if (trainerData[k] && trainerData[k].box === targetBox) {
                        activeQueue.push(item);
                    }
                });
                activeQueue.sort(() => Math.random() - 0.5);
            }

            nextTrainerCardItem();
        }

        function nextTrainerCardItem() {
            document.getElementById('trainer-solution').classList.add('hidden');
            document.getElementById('trainer-hint-box').classList.add('hidden');

            if (activeQueue.length === 0) {
                document.getElementById('trainer-active-area').classList.add('hidden');
                document.getElementById('trainer-empty').classList.remove('hidden');
                return;
            }

            document.getElementById('trainer-active-area').classList.remove('hidden');
            document.getElementById('trainer-empty').classList.add('hidden');

            let currentItem = activeQueue[0];
            currentTrainerItem = currentItem;

            document.getElementById('trainer-progress').innerText = `Noch ${activeQueue.length}`;
            
            let imgSource = currentItem.isCustom ? currentItem.image : (currentItem.data ? currentItem.data.image : '');
            document.getElementById('trainer-img').src = imgSource || '';

            let questionTitle = "";
            let correctAnswer = "";
            let options = [];

            if (currentItem.isCustom) {
                questionTitle = currentItem.title;
                correctAnswer = currentItem.correct;
                options = [...currentItem.options].sort(() => Math.random() - 0.5);
            } else {
                let b = currentItem.data;
                let qType = Math.floor(Math.random() * 3);
                if (qType === 0) {
                    questionTitle = "Welches Bauwerk (mit Ort) ist auf dem Bild zu sehen?";
                    correctAnswer = `${b.name} (${b.location})`;
                } else if (qType === 1) {
                    questionTitle = `Wer ist der Architekt des Bauwerks "${b.name}"?`;
                    correctAnswer = b.architect;
                } else {
                    questionTitle = `In welchem Zeitraum wurde das Bauwerk "${b.name}" errichtet?`;
                    correctAnswer = b.year;
                }

                options = [correctAnswer];
                while (options.length < 5) {
                    let randomB = buildings[Math.floor(Math.random() * buildings.length)];
                    let wrongVal = qType === 0 ? `${randomB.name} (${randomB.location})` : qType === 1 ? randomB.architect : randomB.year;
                    if (!options.includes(wrongVal)) options.push(wrongVal);
                }
                options.sort(() => Math.random() - 0.5);
            }

            document.getElementById('trainer-question-title').innerText = questionTitle;

            let optContainer = document.getElementById('trainer-options');
            optContainer.innerHTML = '';
            options.forEach((opt, idx) => {
                let btn = document.createElement('button');
                btn.className = "w-full text-left p-3 rounded-xl border border-[#e8e2d5] bg-[#f5f2eb]/40 hover:bg-[#f5f2eb] font-medium text-xs transition flex items-center space-x-3";
                btn.innerHTML = `<span class="w-5 h-5 rounded-md bg-[#e8e2d5] text-[#595042] flex items-center justify-center text-[10px] font-bold">${String.fromCharCode(65 + idx)}</span> <span class="text-[#24211a]">${opt}</span>`;
                btn.onclick = () => checkTrainerAnswer(opt === correctAnswer, correctAnswer);
                optContainer.appendChild(btn);
            });
        }

        function showHint() {
            if (!currentTrainerItem) return;
            let hintBox = document.getElementById('trainer-hint-box');
            let hintText = document.getElementById('trainer-hint-text');
            if(currentTrainerItem.isCustom) {
                hintText.innerHTML = `<strong>Tipp:</strong> Besondere Frage!`;
            } else {
                let b = currentTrainerItem.data;
                hintText.innerHTML = `<strong>Tipp:</strong> Stil: <em>${b.style}</em>`;
            }
            hintBox.classList.remove('hidden');
        }

        function checkTrainerAnswer(isCorrect, correctAnswer) {
            let bIdx = currentTrainerItem.index;
            let solBox = document.getElementById('trainer-solution');
            let resTitle = document.getElementById('trainer-result-title');
            let resDetail = document.getElementById('trainer-result-detail');

            solBox.classList.remove('hidden');
            let buttons = document.querySelectorAll('#trainer-options button');
            buttons.forEach(btn => btn.disabled = true);

            if (isCorrect) {
                solBox.className = "mt-4 p-4 rounded-2xl border-l-2 border-emerald-600 bg-emerald-50 text-emerald-900";
                resTitle.innerHTML = `Richtig!`;
                resDetail.innerHTML = `Antwort: <strong>${correctAnswer}</strong>`;
                if (trainerData[bIdx].box < 3) trainerData[bIdx].box += 1;
            } else {
                solBox.className = "mt-4 p-4 rounded-2xl border-l-2 border-red-600 bg-red-50 text-red-900";
                resTitle.innerHTML = `Falsch!`;
                resDetail.innerHTML = `Richtig wäre: <strong>${correctAnswer}</strong> (Zurück in In Wiederholung)`;
                trainerData[bIdx].box = 1;
            }
            saveTrainerData();
            updateTrainerStats();
        }

        function nextTrainerCard() {
            activeQueue.shift();
            nextTrainerCardItem();
        }

        function switchTab(tab) {
            ['summary', 'cards', 'trainer', 'sim', 'exam', 'theory'].forEach(t => {
                let sec = document.getElementById('sec-' + t);
                let tabEl = document.getElementById('tab-' + t);
                if (sec) sec.classList.add('hidden');
                if (tabEl) tabEl.className = "flex-grow md:flex-grow-0 px-3 py-1.5 rounded-xl md:rounded-full text-xs font-medium transition text-[#736856] hover:text-[#24211a] whitespace-nowrap text-center";
            });
            let targetSec = document.getElementById('sec-' + tab);
            let targetTab = document.getElementById('tab-' + tab);
            if (targetSec) targetSec.classList.remove('hidden');
            if (targetTab) targetTab.className = "flex-grow md:flex-grow-0 px-3 py-1.5 rounded-xl md:rounded-full text-xs font-medium transition bg-white text-[#24211a] shadow-sm whitespace-nowrap text-center";
        }

        }
